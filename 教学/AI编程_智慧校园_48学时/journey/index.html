先准备，再做第一张页面

这条主线只用一个工程：桌面 campus-ai-lab/app。前端业务页面逐站增加；认证、请求、数据库和完整后端服务由教师提供可读源码。你会让CodeBuddy 写页面，也会读和修改后端业务规则。参考源码是失败时的对照，不是你的完成记录。

先装工具，再复制工程

  1. 完整解压教学包，打开 resources/project-journey。
  2. 把其中的 app 复制到桌面 campus-ai-lab。若已有同名作业，保留它，新建 campus-ai-lab-v12；不要覆盖。

function openDevice(id: string) { location.hash = catalogHash({ ...filter, deviceId: id }) }

  selectedId.value = ''

}
function clearFilters() {
  Object.assign(filter, { keyword: '', online: 'all', spaceId: '', category: '' })
}

onBeforeUnmount(() => window.removeEventListener('hashchange', hashChanged))

import { getLocations, createCampus, createBuilding, bindSpace } from '../api/full'

import SourceBanner from '../components/SourceBanner.vue'

const store = useCampusStore(),
  read = useRequestScope(),
  write = useRequestScope()
const data = ref<Locations | null>(null),
  spaces = ref<Space[]>([])
const campusName = ref(''),
  campusId = ref(''),
  buildingName = ref(''),
  spaceId = ref(''),
  buildingId = ref(''),
  floor = ref(''),
  notice = ref('')
function refresh() {
  return read.run(
    async (signal) => ({ locations: await getLocations(signal), spaces: await getSpaces() }),
    (result) => {
      data.value = result.locations
      spaces.value = result.spaces.items
    },
    (error) => {
      if (error instanceof ApiError && error.status === 403) {
        data.value = null
        spaces.value = []
      }
    },
  )
}
async function save(kind: 'campus' | 'building' | 'binding') {
  if (write.busy.value || store.capabilities?.canAdmin !== true) return
  notice.value = ''
  await write.run(
    (signal) =>
      kind === 'campus'
        ? createCampus(campusName.value.trim(), store.csrf, signal)
        : kind === 'building'
          ? createBuilding(campusId.value, buildingName.value.trim(), store.csrf, signal)
          : bindSpace(
              spaceId.value,
              buildingId.value,
              floor.value.trim() || null,
              store.csrf,
              signal,
            ),
    () => {
      notice.value = '服务端已确认保存。'
      void refresh()
    },
  )
}
function bindingText(id: string) {
  const binding = data.value?.bindings.find((b) => b.spaceId === id)
  const building = data.value?.buildings.find((b) => b.id === binding?.buildingId)
  const campus = data.value?.campuses.find((c) => c.id === building?.campusId)
  return binding
    ? `${campus?.name ?? '未知校区'} / ${building?.name ?? '未知楼栋'} / ${binding.floor ?? '楼层未填'}`
    : '未绑定'
}
onMounted(refresh)

  data = ref<Overview | null>(null)

  return scope.run(
    getOverview,
    (value) => {
      data.value = value

  )

  data.value?.devices.find((d) => d.id === deviceId)?.tags.find((t) => t.id === tagId)?.name ??
  tagId

<script setup lang="ts">
import { computed, onMounted, onBeforeUnmount, ref, watch, defineAsyncComponent } from 'vue'
import { useCampusStore } from '../stores/campus'
import { useRequestScope } from '../composables/useRequestScope'
import { getHistory } from '../api/full'
import type { HistoryResult } from '../types/full'
import { timeText } from '../domain/course'
import { historyHash } from '../domain/read-models'
import { ApiError } from '../api/http'
import SourceBanner from '../components/SourceBanner.vue'
import ReadStatus from '../components/ReadStatus.vue'
const HistoryChart = defineAsyncComponent(() => import('../components/HistoryChart.vue'))
const store = useCampusStore(),
  scope = useRequestScope()
const deviceId = ref(''),
  tagId = ref(''),
  minutes = ref(60),
  data = ref<HistoryResult | null>(null)
const linkError = ref(''), queriedAt = ref<number | null>(null)
const tags = computed(() => store.devices.find((d) => d.id === deviceId.value)?.tags ?? [])
watch(deviceId, () => {
  tagId.value = tags.value[0]?.id ?? ''
}, { flush: 'sync' })
async function query() {
  if (!deviceId.value || !tagId.value || store.observedAt === null) return
  linkError.value = ''
  const requestedDevice = deviceId.value, requestedTag = tagId.value
  history.replaceState(null, '', historyHash(requestedDevice, requestedTag))
  const startedAt = Date.now()
  const params = new URLSearchParams({
    deviceId: deviceId.value,
    tagId: tagId.value,
    from: String(store.observedAt - minutes.value * 60000),
    to: String(store.observedAt),
  })
  await scope.run(
    (signal) => getHistory(params, signal),
    (value) => {
      if (value.deviceId !== requestedDevice || value.tagId !== requestedTag) throw new ApiError('历史响应点位不匹配。')
      data.value = value
      queriedAt.value = value.queriedAt ?? startedAt
    },
    (error) => {
      if (error instanceof ApiError && error.status === 403) data.value = null
    },
  )
}
onMounted(async () => {
  await store.refreshDevices()
  if (!scope.active()) return
  readLocation()
})
function readLocation() {

<script setup lang="ts">
import { journeyStage, closurePanel } from '../journey/features'
import { computed, onMounted, onBeforeUnmount, ref } from 'vue'
import { useCampusStore } from '../stores/campus'
import { useRequestScope } from '../composables/useRequestScope'
import {
  getCases,
  getCase,
  getHandlers,
  getDemoStatus,
  triggerDemo,
  actOnCase,
  type CaseAction,
} from '../api/full'
import { ApiError, safeError } from '../api/http'
import type { CaseList, AlarmDetail, Handler, DemoStatus } from '../types/full'
import {
  caseQuery,
  changeCaseState,
  validCaseState,
  stateNames,
  sourceStateNames,
  actionNames,
  closeBlockers,
  timeText,
} from '../domain/course'
import SourceBanner from '../components/SourceBanner.vue'
import ReadStatus from '../components/ReadStatus.vue'
const store = useCampusStore(),
  listScope = useRequestScope(),
  detailScope = useRequestScope(),
  support = useRequestScope(),
  write = useRequestScope()
const state = ref(validCaseState(new URLSearchParams(location.hash.split('?')[1]).get('state'))),
  skip = ref(0),
  limit = ref(10),
  handlerFilter = ref('')
const list = ref<CaseList | null>(null),
  detail = ref<AlarmDetail | null>(null),
  handlers = ref<Handler[]>([]),
  demo = ref<DemoStatus | null>(null)
const selectedId = ref(''),
  note = ref(''),
  assigneeUid = ref(''),
  disposition = ref(''),
  scenario = ref(''),
  notice = ref(''),
  filterError = ref('')
const conflict = ref(false),
  latestReady = ref(false),
  readDenied = ref(false)
function clearDeniedResources() {
  readDenied.value = true
  listScope.cancel()
  detailScope.cancel()
  support.cancel()
  write.cancel()
  list.value = null
  detail.value = null
  handlers.value = []
  demo.value = null
  selectedId.value = ''
  note.value = ''
  disposition.value = ''
  assigneeUid.value = ''
  conflict.value = false
  latestReady.value = false
  write.error.value = ''
  notice.value = ''
}
const canHandle = computed(
  () =>
    !readDenied.value && store.capabilities?.canHandleAlarms === true &&
    (store.capabilities?.canAdmin === true || detail.value?.assigneeUid === store.user?.uid),
)
const pageCounts = computed(() => Object.keys(stateNames).map(state => ({ state, count: list.value?.items.filter(item => item.state === state).length ?? 0 })))
const isClosed = computed(() => detail.value?.state === 'CLOSED')
const blocked = computed(() => (detail.value ? closeBlockers(detail.value) : []))
const operationBlocked = computed(
  () =>
    write.busy.value || detailScope.busy.value || conflict.value || !detail.value || isClosed.value,
)
function refresh() {
  if (readDenied.value) return Promise.resolve(false)
  return listScope.run(
    (signal) =>
      getCases(
        caseQuery(state.value, skip.value, limit.value, handlerFilter.value === '@me' ? store.user!.uid : handlerFilter.value),
        signal,
      ),
    (value) => {
      list.value = value
    },
    (error) => {
      if (error instanceof ApiError && error.status === 403) clearDeniedResources()
    },
  )
}
function readDetail() {
  const id = selectedId.value
  if (!id || readDenied.value) return Promise.resolve(false)
  return detailScope.run(
    (signal) => getCase(id, signal),
    (value) => {
      if (selectedId.value !== id) return
      detail.value = value
      if (conflict.value) latestReady.value = true
    },
    (error) => {
      if (error instanceof ApiError && error.status === 403) clearDeniedResources()
    },
  )
}
function selectCase(id: string) {
  if (write.busy.value) return
  detailScope.cancel()
  selectedId.value = id
  detail.value = null
  note.value = ''
  disposition.value = ''
  assigneeUid.value = ''
  conflict.value = false
  latestReady.value = false
  notice.value = ''
  write.error.value = ''
  void readDetail()
}
function updateState(value: string) {
  try {
    const next = changeCaseState(value)
    state.value = next.state
    skip.value = next.skip
    filterError.value = ''
    const query = state.value ? `?state=${state.value}` : ''
    history.replaceState(null, '', `#cases${query}`)
    void refresh()
  } catch (error) {
    filterError.value = safeError(error)
  }
}
function hashChanged() {
  if (location.hash.split('?')[0] === '#cases')
    updateState(validCaseState(new URLSearchParams(location.hash.split('?')[1]).get('state')))
}
function otherFilter() {
  skip.value = 0
  void refresh()
}
function page(delta: number) {
  skip.value = Math.max(0, skip.value + delta * limit.value)
  void refresh()
}
async function trigger() {
  if (readDenied.value || !scenario.value || write.busy.value || store.capabilities?.canRunDemoScenarios !== true)
    return
  // 新建操作接管详情；旧事件的迟到成功、403 和错误都不得影响新事件。
  detailScope.cancel()
  detailScope.error.value = ''
  notice.value = ''
  await write.run(
    (signal) => triggerDemo(scenario.value, store.csrf, signal),
    (value) => {
      detailScope.cancel()
      detailScope.error.value = ''
      selectedId.value = value.id
      detail.value = value
      conflict.value = false
      latestReady.value = false
      note.value = ''
      disposition.value = ''
      assigneeUid.value = ''
      notice.value = '已创建本地模拟事件。'
      skip.value = 0
      void refresh()
    },
  )
}
async function act(action: CaseAction) {
  if (journeyStage < 9 && ['recover', 'verify-recovery', 'close'].includes(action)) return
  if (operationBlocked.value || !detail.value) return
  if (
    (action === 'assign' && store.capabilities?.canAssignAlarms !== true) ||
    (action === 'recover' && store.capabilities?.canRunDemoScenarios !== true) ||
    (!['assign', 'recover'].includes(action) && !canHandle.value)
  )
    return
  const current = detail.value,
    body: Record<string, unknown> = { expectedVersion: current.version }
  if (action === 'assign') {
    if (!assigneeUid.value || !note.value.trim()) return
    body.assigneeUid = assigneeUid.value
    body.note = note.value.trim()
  }
  if (action === 'start' || action === 'notes') {
    if (!note.value.trim()) return
    body.note = note.value.trim()
  }
  if (action === 'close') {
    if (!disposition.value.trim() || blocked.value.length) return
    body.disposition = disposition.value.trim()
  }
  detailScope.cancel()
  notice.value = ''
  await write.run(
    (signal) => actOnCase(current.id, action, body, store.csrf, signal),
    (value) => {
      detail.value = value
      notice.value = '服务端已确认操作，版本与时间线已更新。'
      note.value = ''
      if (action === 'close') disposition.value = ''
      if (['recover', 'verify-recovery', 'close'].includes(action)) void readDetail()
      void refresh()
    },
    (error) => {
      if (error instanceof ApiError && error.status === 409) {
        conflict.value = true
        latestReady.value = false
        void readDetail()
      }
      // 输入保持不变；403 不能自动换身份或重试，500 不能宣称事务已回滚。
    },
  )
}
function acknowledge() {
  if (latestReady.value) {
    conflict.value = false
    latestReady.value = false
    write.error.value = ''
    notice.value = '已重新核对当前版本；请明确选择下一次操作。'
  }
}
onMounted(() => {
  void refresh()
  void support.run(
    async (signal) => ({ handlers: await getHandlers(signal), demo: await getDemoStatus(signal) }),
    (value) => {
      handlers.value = value.handlers.items
      demo.value = value.demo
      scenario.value = value.demo.scenarios[0]?.id ?? ''
    },
  )
  window.addEventListener('hashchange', hashChanged)
})
onBeforeUnmount(() => window.removeEventListener('hashchange', hashChanged))
</script>

  deviceId.value = ''
  ownerUid.value = ''
  note.value = ''
  notice.value = ''
  conflict.value = false
  latestReady.value = false
  write.error.value = ''
}
function refresh() {
  if (readDenied.value) return Promise.resolve(false)
  return listScope.run(
    async (signal) => ({ owners: await getOwners(signal), handlers: await getHandlers(signal) }),
    (value) => {
      owners.value = value.owners.items
      handlers.value = value.handlers.items
    },
    (error) => {
      if (error instanceof ApiError && error.status === 403) clearDeniedResources()
    },
  )
}
function readDetail() {
  if (!deviceId.value || readDenied.value) return Promise.resolve(false)
  const selected = deviceId.value
  return detailScope.run(
    (signal) => getOwner(selected, signal),
    (value) => {
      detail.value = value
      if (conflict.value) latestReady.value = true
    },
    (error) => {
      if (error instanceof ApiError && error.status === 403) clearDeniedResources()
    },
  )
}
function select(id: string) {
  if (write.busy.value) return
  detailScope.cancel()
  deviceId.value = id
  detail.value = null
  ownerUid.value = ''
  note.value = ''
  notice.value = ''
  conflict.value = false
  latestReady.value = false
  write.error.value = ''
  void readDetail()
}
async function save() {
  if (
    readDenied.value || !detail.value ||
    !ownerUid.value ||
    !note.value.trim() ||
    write.busy.value ||
    detailScope.busy.value ||
    conflict.value ||
    store.capabilities?.canAdmin !== true

      saveOwner(
        original.deviceId,
        ownerUid.value,
        note.value.trim(),
        original.version,
        store.csrf,
        signal,
      ),

      detail.value = value
      note.value = ''

        conflict.value = true
        latestReady.value = false
        void readDetail()
      }
    },
  )
}

  conflict.value = false

  write.error.value = ''
}
